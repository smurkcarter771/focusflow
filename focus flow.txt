    <!-- JavaScript Logic -->
    <script>
        let totalSeconds = 25 * 60;
        let timeLeft = 25 * 60;
        let timerId = null;
        let isRunning = false;
        let currentMode = 'work';

        const quotes = [
            "\"The secret of getting ahead is getting started.\"",
            "\"Focus is a muscle. The more you practice, the stronger it gets.\"",
            "\"Don't watch the clock; do what it does. Keep going.\"",
            "\"Small disciplines repeated with consistency every day lead to great achievements.\""
        ];

        // Web Audio API generator for clean ambient sounds without external files
        let audioCtx = null;
        let noiseNode = null;
        let noiseGain = null;
        let activeSoundType = null;

        function switchMode(mode) {
            pauseTimer();
            currentMode = mode;
            
            const btnWork = document.getElementById('btnWork');
            const btnShort = document.getElementById('btnShortBreak');
            const btnLong = document.getElementById('btnLongBreak');
            const progressRing = document.getElementById('timerProgress');
            const cardGlow = document.getElementById('cardGlow');
            const statusLabel = document.getElementById('sessionStatusLabel');

            // Reset active styles
            btnWork.className = "py-2.5 text-xs font-semibold rounded-xl transition-all text-slate-400 hover:text-white";
            btnShort.className = "py-2.5 text-xs font-semibold rounded-xl transition-all text-slate-400 hover:text-white";
            btnLong.className = "py-2.5 text-xs font-semibold rounded-xl transition-all text-slate-400 hover:text-white";

            if (mode === 'work') {
                totalSeconds = 25 * 60;
                btnWork.className = "py-2.5 text-xs font-bold rounded-xl transition-all bg-rose-600 text-white shadow-lg shadow-rose-600/20";
                progressRing.className = "text-rose-500 timer-ring";
                cardGlow.className = "absolute -top-24 -right-24 w-48 h-48 bg-rose-500/10 rounded-full blur-3xl pointer-events-none transition-all duration-500";
                statusLabel.innerText = "Ready to Focus";
                statusLabel.className = "text-xs font-semibold text-rose-400 uppercase tracking-widest mt-1";
            } else if (mode === 'shortBreak') {
                totalSeconds = 5 * 60;
                btnShort.className = "py-2.5 text-xs font-bold rounded-xl transition-all bg-emerald-600 text-white shadow-lg shadow-emerald-600/20";
                progressRing.className = "text-emerald-500 timer-ring";
                cardGlow.className = "absolute -top-24 -right-24 w-48 h-48 bg-emerald-500/10 rounded-full blur-3xl pointer-events-none transition-all duration-500";
                statusLabel.innerText = "Take a Breath";
                statusLabel.className = "text-xs font-semibold text-emerald-400 uppercase tracking-widest mt-1";
            } else if (mode === 'longBreak') {
                totalSeconds = 15 * 60;
                btnLong.className = "py-2.5 text-xs font-bold rounded-xl transition-all bg-sky-600 text-white shadow-lg shadow-sky-600/20";
                progressRing.className = "text-sky-500 timer-ring";
                cardGlow.className = "absolute -top-24 -right-24 w-48 h-48 bg-sky-500/10 rounded-full blur-3xl pointer-events-none transition-all duration-500";
                statusLabel.innerText = "Rest & Recharge";
                statusLabel.className = "text-xs font-semibold text-sky-400 uppercase tracking-widest mt-1";
            }

            timeLeft = totalSeconds;
            updateDisplay();
        }

        function toggleTimer() {
            if (isRunning) {
                pauseTimer();
            } else {
                startTimer();
            }
        }

        function startTimer() {
            if (isRunning) return;
            isRunning = true;
            document.getElementById('mainToggleText').innerText = "Pause Session";
            document.getElementById('mainToggleIcon').className = "fa-solid fa-pause";
            document.getElementById('sessionStatusLabel').innerText = currentMode === 'work' ? "Deep Focus Active" : "Break Time";

            // Rotate quote
            const randomQuote = quotes[Math.floor(Math.random() * quotes.length)];
            document.getElementById('quoteText').innerText = randomQuote;

            timerId = setInterval(() => {
                if (timeLeft > 0) {
                    timeLeft--;
                    updateDisplay();
                } else {
                    pauseTimer();
                    alert("Session completed! Great work.");
                    resetTimer();
                }
            }, 1000);
        }

        function pauseTimer() {
            isRunning = false;
            clearInterval(timerId);
            document.getElementById('mainToggleText').innerText = "Resume Session";
            document.getElementById('mainToggleIcon').className = "fa-solid fa-play";
            document.getElementById('sessionStatusLabel').innerText = "Paused";
        }

        function resetTimer() {
            pauseTimer();
            timeLeft = totalSeconds;
            document.getElementById('mainToggleText').innerText = "Start Session";
            document.getElementById('mainToggleIcon').className = "fa-solid fa-play";
            document.getElementById('sessionStatusLabel').innerText = currentMode === 'work' ? "Ready to Focus" : "Break Ready";
            updateDisplay();
        }

        function updateDisplay() {
            const minutes = Math.floor(timeLeft / 60);
            const seconds = timeLeft % 60;
            const formatted = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
            document.getElementById('timeDisplay').innerText = formatted;
            document.title = `${formatted} - FocusFlow`;

            // Update SVG circle stroke offset
            const circumference = 615.7;
            const progressOffset = circumference - (timeLeft / totalSeconds) * circumference;
            document.getElementById('timerProgress').style.strokeDashoffset = progressOffset;
        }

        // Ambient Sound generator logic using Web Audio API
        function toggleAmbient(type) {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }

            if (activeSoundType === type) {
                stopAmbient();
                return;
            }

            stopAmbient();
            activeSoundType = type;
            document.getElementById('audioStatus').innerText = "Playing (" + type + ")";

            // Create pink/brown noise buffer for natural ambient sound
            const bufferSize = audioCtx.sampleRate * 2;
            const noiseBuffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
            const output = noiseBuffer.getChannelData(0);
            
            let b0 = 0, b1 = 0, b2 = 0, b3 = 0, b4 = 0, b5 = 0, b6 = 0;
            for (let i = 0; i < bufferSize; i++) {
                const white = Math.random() * 2 - 1;
                if (type === 'rain') {
                    // Filtered noise for rain-like sound
                    b0 = 0.99886 * b0 + white * 0.0555179;
                    output[i] = b0 * 0.3;
                } else {
                    // Brown noise for deep focus hum
                    b0 = 0.997 * b0 + white * 0.05;
                    output[i] = b0 * 0.4;
                }
            }

            noiseNode = audioCtx.createBufferSource();
            noiseNode.buffer = noiseBuffer;
            noiseNode.loop = true;

            noiseGain = audioCtx.createGain();
            noiseGain.gain.value = 0.15; // comfortable background volume

            noiseNode.connect(noiseGain);
            noiseGain.connect(audioCtx.destination);
            noiseNode.start(0);

            // Button styling indicator
            document.getElementById('btnRain').classList.remove('border-rose-500');
            document.getElementById('btnBrownNoise').classList.remove('border-rose-500');
            if (type === 'rain') document.getElementById('btnRain').classList.add('border-rose-500');
            if (type === 'brownNoise') document.getElementById('btnBrownNoise').classList.add('border-rose-500');
        }

        function stopAmbient() {
            if (noiseNode) {
                try { noiseNode.stop(); } catch(e) {}
                noiseNode.disconnect();
                noiseNode = null;
            }
            activeSoundType = null;
            document.getElementById('audioStatus').innerText = "Muted";
            document.getElementById('btnRain').classList.remove('border-rose-500');
            document.getElementById('btnBrownNoise').classList.remove('border-rose-500');
        }
    </script>
